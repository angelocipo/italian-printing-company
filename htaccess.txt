# Rename to .htaccess and upload to the site root (Apache hosting).
# If a .htaccess already exists, paste these lines at the top of it.
RewriteEngine On

# 1. Merge the two sites: non-www -> www
RewriteCond %{HTTP_HOST} ^italianprintingcompany\.com$ [NC]
RewriteRule ^(.*)$ https://www.italianprintingcompany.com/$1 [R=301,L]

# 2. Force https
RewriteCond %{HTTPS} off
RewriteRule ^(.*)$ https://www.italianprintingcompany.com/$1 [R=301,L]

# 3. /index.html -> /
RewriteCond %{THE_REQUEST} \s/index\.html[\s?] [NC]
RewriteRule ^index\.html$ / [R=301,L]
